// Display SQL using DuckDB WASM with query based on inputs
let query = `
SELECT * FROM brfss 
WHERE SEXVAR IN (null, ${[gender]}) 
AND (AGE = '${age}' OR '${age}' = '')
AND MARITAL IN (null, ${[maritalStatus]})
AND EDUCA IN (null, ${[educationLevel]})
AND PRIMINSR IN (null, ${[primSourceIns]})
AND PERSDOC3 IN (null, ${[persHealthProv]})
AND MEDCOST1 IN (null, ${[cantAffordDoctor]})
AND CHECKUP1 IN (null, ${[lengthSinceLastCheckup]})
AND CVDINFR4 IN (null, ${[heartAttack]})
AND CVDCRHD4 IN (null, ${[anginaOrCoronary]})
AND CVDSTRK3 IN (null, ${[stroke]})
AND ASTHMA3 IN (null, ${[asthma]})
AND CHCSCNC1 IN (null, ${[skinCancer]})
AND CHCOCNC1 IN (null, ${[otherCancer]})
AND OH8_1 IN (NULL, ${[TBIq1]})
AND OH8_2 IN (NULL, ${[TBIq2]})
AND OH8_3 IN (NULL, ${[TBIq3]})
AND OH8_4 IN (NULL, ${[TBIq4]})
AND OH8_5 IN (NULL, ${[TBIq5]})
AND OH8_6 IN (NULL, ${[TBIq6]})
AND OH8_7 IN (NULL, ${[TBIq7]})
AND (OH8_8 = '${TBIq8}' OR '${TBIq8}' = '')
AND (OH8_9 = '${TBIq9}' OR '${TBIq9}' = '')
AND OH8_10 IN (NULL, ${[TBIq10]})
AND OH8_11 IN (NULL, ${[TBIq11]})
`

display(Inputs.table(await db.sql([query])));
